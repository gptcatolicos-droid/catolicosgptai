// === ADMIN MENU CONTROLS ===
const MENU_CONFIG_FILE = path.join(DATA_DIR, 'menu-config.json');
const DEFAULT_MENU_CONFIG = {
  chatIA: true,
  santoDelDia: true,
  oracionDelDia: true,
  santoral: true,
  infografias: true,
  oraciones: true,
  ninos: true,
  videos: true,
  podcasts: true,
  blog: true,
  catequesisIA: true,
  misas: true
};

function getMenuConfig() {
  try {
    if (!fs.existsSync(MENU_CONFIG_FILE)) return { ...DEFAULT_MENU_CONFIG };
    const saved = JSON.parse(fs.readFileSync(MENU_CONFIG_FILE, 'utf8'));
    return { ...DEFAULT_MENU_CONFIG, ...saved };
  } catch (err) {
    console.error('[Menu Config] No se pudo leer la configuración:', err.message);
    return { ...DEFAULT_MENU_CONFIG };
  }
}

function saveMenuConfig(nextConfig = {}) {
  const normalized = Object.keys(DEFAULT_MENU_CONFIG).reduce((acc, key) => {
    acc[key] = nextConfig[key] === true;
    return acc;
  }, {});
  fs.writeFileSync(MENU_CONFIG_FILE, JSON.stringify(normalized, null, 2), 'utf8');
  return normalized;
}

function getMenuVisibilityCss() {
  const config = getMenuConfig();
  const disabled = Object.entries(config)
    .filter(([, enabled]) => !enabled)
    .map(([key]) => `[data-menu-key="${key}"]{display:none!important}`)
    .join('');
  return `<style id="admin-menu-visibility">${disabled}</style>`;
}
